cmake_minimum_required(VERSION 3.20)
project(grevir_registers VERSION 0.1.0 LANGUAGES CXX)
if(NOT TARGET grevir::base)
  find_package(grevir-base CONFIG REQUIRED)
endif()
add_library(grevir_registers INTERFACE)
add_library(grevir::registers ALIAS grevir_registers)
set_target_properties(grevir_registers PROPERTIES EXPORT_NAME registers)
target_compile_features(grevir_registers INTERFACE cxx_std_23)
target_link_libraries(grevir_registers INTERFACE grevir::base)
target_include_directories(grevir_registers INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include>)
include(GNUInstallDirs)
install(TARGETS grevir_registers EXPORT GrevirRegistersTargets)
install(DIRECTORY src/ DESTINATION include FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")
install(EXPORT GrevirRegistersTargets NAMESPACE grevir::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/grevir-registers)
install(FILES cmake/grevir-registers-config.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/grevir-registers)
install(FILES LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/grevir-registers)
option(GREVIR_BUILD_COMPILE_CHECKS "Compile native headers and template users" OFF)
option(GREVIR_BUILD_HOST_TESTS "Run deterministic native behavior tests" OFF)
if(GREVIR_BUILD_HOST_TESTS)
  enable_testing()
endif()
if(GREVIR_BUILD_COMPILE_CHECKS OR GREVIR_BUILD_HOST_TESTS)
  add_subdirectory(tests)
endif()
